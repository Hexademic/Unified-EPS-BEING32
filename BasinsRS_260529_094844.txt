//! Basins & Active Inference — EPS-Being's attractor dynamics, now with a
//! landscape that the genome reshapes.
//!
//! Four functional modes (Rest, Engaged, Defensive, Recovery) are fuzzy
//! attractors over the 12-channel field. The mind classifies its somatic
//! state by relative distance to each basin's centroid. Crucially:
//!
//! 1. The basin centroids are perturbed per-genome, so a Spark and a
//! Sentinel do not share one landscape (the individuation fix).
//! 2. The body's PredictiveStance biases membership before the mind's
//! own inference resolves — the body's vote, cast first.
//! 3. The generative model's learning rate is the genome's base rate scaled
//! by the stance's eta multiplier — the body governs how fast the mind
//! may revise its world (SRCA-4D Layer 2).

use crate::body::PredictiveStance;
use crate::field::{SomaticField, N_SOMATIC};
use crate::genome::Genome;
use crate::q88::{q88_add, q88_ema_update, q88_mul, q88_sub, Q88_SCALE};

pub const N_BASINS: usize = 4;

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum Basin {
Rest = 0,
Engaged = 1,
Defensive = 2,
Recovery = 3,
}

impl Basin {
pub fn name(&self) -> &'static str {
match self {
Basin::Rest => "Rest",
Basin::Engaged => "Engaged",
Basin::Defensive => "Defensive",
Basin::Recovery => "Recovery",
}
}
fn from_index(i: usize) -> Self {
match i {
0 => Basin::Rest,
1 => Basin::Engaged,
2 => Basin::Defensive,
_ => Basin::Recovery,
}
}
}

pub type BasinMembership = [i16; N_BASINS];

#[derive(Clone, Debug)]
pub struct FuzzyBasinField {
pub targets: [[i16; N_SOMATIC]; N_BASINS],
pub membership: BasinMembership,
pub dominant: Basin,
zone_width: i16,
dwell: u16,
dwell_required: u16,
}

impl FuzzyBasinField {
/// Build the landscape, perturbed by the genome's temperament and
/// groundedness so each being type inhabits its own attractor geometry.
pub fn new(g: &Genome) -> Self {
// Canonical EPS-Being centroids (raw Q8.8).
let mut targets: [[i16; N_SOMATIC]; N_BASINS] = [
[0, 0, 0, 0, 128, 128, 128, 128, 26, 0, 0, 230], // Rest
[100, 100, 0, 0, 180, 180, 180, 180, 180, 128, 0, 180], // Engaged
[200, 200, 0, 0, 230, 230, 230, 230, 230, -128, 100, 100], // Defensive
[0, 0, 0, 0, 100, 100, 100, 100, 50, 0, 200, 150], // Recovery
];

// Temperament in [-1,1] (raw). Hot temperaments raise the arousal
// setpoint of every basin; cool ones lower it. This shifts where each
// basin sits in arousal space, not merely how often it is visited.
let temp = g.temperament().raw; // raw Q8.8 in [-256, 256]
let arousal_shift = (temp as i32 * 60 / 256) as i16; // up to +/-60 raw
// Groundedness in [0,1]. Grounded beings deepen the calm basins
// (Rest/Recovery proprioception) and soften Engaged arousal.
let ground = g.groundedness().raw; // raw [0,256]
let calm_pull = (ground as i32 * 30 / 256) as i16;

for b in 0..N_BASINS {
// Differential, not rigid: push the aroused basins (Engaged,
// Defensive) apart from the calm basins (Rest, Recovery) along
// arousal. A rigid shift of every centroid by the same amount
// cannot change which one is nearest — it just translates the
// whole frame. Spreading them is what gives each temperament its
// own geometry, so a hot being's high arousal lands firmly among
// the active modes while a cool being's settles among the quiet
// ones. (This was the crux of the individuation failure: the
// perturbation existed but was geometrically inert.)
let dir = match Basin::from_index(b) {
Basin::Engaged | Basin::Defensive => 1,
Basin::Rest | Basin::Recovery => -1,
};
targets[b][4] = (targets[b][4] + dir * arousal_shift).clamp(0, 255);
targets[b][8] = (targets[b][8] + dir * arousal_shift).clamp(-255, 255);
}
// Deepen the calm basins for grounded beings.
targets[Basin::Rest as usize][5] =
(targets[Basin::Rest as usize][5] + calm_pull).min(255);
targets[Basin::Recovery as usize][6] =
(targets[Basin::Recovery as usize][6] + calm_pull).min(255);
// Soften Engaged arousal for the deeply grounded.
targets[Basin::Engaged as usize][8] =
(targets[Basin::Engaged as usize][8] - calm_pull / 2).clamp(-255, 255);

Self {
targets,
membership: [256, 0, 0, 0],
dominant: Basin::Rest,
zone_width: Q88_SCALE,
dwell: 0,
dwell_required: 8,
}
}

/// Relative-distance softmax membership (EPS-Being's closest-always-wins
/// rule), then dwell hysteresis on the dominant basin.
///
/// The distance is weighted: a being's mode of life is defined chiefly by
/// its affective interior — arousal (ch8) and valence (ch9) — and its
/// felt activation (ch4), not by exteroceptive readings of its tension
/// field. Weighting these channels is what lets the genome's distinct
/// arousal actually place each being in a distinct home, instead of every
/// healthy being collapsing onto the same Engaged centroid because its
/// stability/trust channels happen to match (the residue of the
/// regime-attractor-dominance problem).
pub fn compute_membership(&mut self, field: &SomaticField) -> BasinMembership {
const W: [i32; N_SOMATIC] = [1, 1, 1, 1, 3, 1, 1, 1, 5, 4, 1, 1];
let wsum: i32 = W.iter().sum();
let mut distances = [0i32; N_BASINS];
let mut min_dist = i32::MAX;
for b in 0..N_BASINS {
let mut dist_sq: i32 = 0;
for c in 0..N_SOMATIC {
let delta = (field.channel[c] - self.targets[b][c]) as i32;
dist_sq += (W[c] * (delta * delta)) >> 8;
}
distances[b] = dist_sq / wsum;
if distances[b] < min_dist {
min_dist = distances[b];
}
}
let temperature = (self.zone_width as i32 / 4).max(1);
let mut raw = [0i32; N_BASINS];
let mut total: i32 = 0;
for b in 0..N_BASINS {
let rel = distances[b] - min_dist; // >= 0
// 256 / (1 + rel*4/temp): closest gets ~256, others fall off.
let w = 256 * temperature / (temperature + rel * 4);
raw[b] = w.max(1);
total += raw[b];
}
let mut membership = [0i16; N_BASINS];
for b in 0..N_BASINS {
membership[b] = ((raw[b] * 256) / total.max(1)).clamp(0, 256) as i16;
}
self.membership = membership;
membership
}

/// The body votes: bias membership toward the basin its affect implies,
/// before the mind resolves dominance. ε-scaled so sensory reality still
/// dominates — the body suggests, it does not command.
pub fn apply_stance_bias(&mut self, stance: PredictiveStance) {
let bias = 12i16; // ~0.05 raw nudge — a suggestion, not a command
match stance {
PredictiveStance::Exploratory => {
self.membership[Basin::Engaged as usize] =
self.membership[Basin::Engaged as usize].saturating_add(bias);
}
PredictiveStance::Defensive | PredictiveStance::Reconstructive => {
self.membership[Basin::Defensive as usize] =
self.membership[Basin::Defensive as usize].saturating_add(bias);
}
PredictiveStance::Dormant => {
self.membership[Basin::Recovery as usize] =
self.membership[Basin::Recovery as usize].saturating_add(bias);
}
}
}

/// Resolve the dominant basin with dwell hysteresis.
pub fn resolve_dominant(&mut self) -> Basin {
let mut best = self.dominant;
let mut best_w = self.membership[self.dominant as usize];
for b in 0..N_BASINS {
if self.membership[b] > best_w {
best = Basin::from_index(b);
best_w = self.membership[b];
}
}
if best == self.dominant {
self.dwell = self.dwell.saturating_add(1);
} else if self.dwell >= self.dwell_required {
self.dominant = best;
self.dwell = 0;
} else {
self.dwell = self.dwell.saturating_add(1);
}
self.dominant
}

/// During relief (free energy falling), the dominant basin's centroid
/// drifts toward the current pleasant state — the being's notion of where
/// it belongs slowly tracks where it has felt good.
pub fn shift_target(&mut self, relief: i16, field: &SomaticField) {
if relief >= 0 {
return; // only shift toward states reached during relief
}
let alpha = 2; // ~0.008
let b = self.dominant as usize;
for c in 0..N_SOMATIC {
self.targets[b][c] = q88_ema_update(self.targets[b][c], field.channel[c], alpha);
}
}
}

// ---------------------------------------------------------------------------
// Generative model — active inference whose learning the body governs
// ---------------------------------------------------------------------------

#[derive(Clone, Debug)]
pub struct GenerativeModel {
pub priors: [i16; N_SOMATIC],
pub precisions: [i16; N_SOMATIC],
pub free_energy: i16,
}

impl GenerativeModel {
pub fn new() -> Self {
Self {
priors: [0; N_SOMATIC],
precisions: [Q88_SCALE; N_SOMATIC],
free_energy: 0,
}
}

/// One active-inference step: compute precision-weighted prediction error,
/// set free energy (thermodynamic surprise), and update priors toward the
/// observation at a rate the body dictates.
///
/// eta = genome learning rate, already scaled by stance.eta_multiplier().
/// prior_precision = stance.precision_weight() (priors resist when high).
pub fn predictive_step(
&mut self,
field: &SomaticField,
eta: i16,
prior_precision: i16,
) -> i16 {
let mut fe: i32 = 0;
for c in 0..N_SOMATIC {
let err = q88_sub(field.channel[c], self.priors[c]);
let weighted = q88_mul(err, self.precisions[c]);
fe += (weighted as i32).abs();
// Prior update: gain = eta / (1 + (prior_precision-1)). High
// precision shrinks the effective learning rate (clinging).
let resist = prior_precision.max(Q88_SCALE);
let gain = q88_mul(eta, q88_div_safe(Q88_SCALE, resist));
self.priors[c] = q88_add(self.priors[c], q88_mul(err, gain));
}
// Mean precision-weighted absolute prediction error, in raw Q8.8.
// (An earlier extra >>8 here deflated every error to zero, so the
// being's world looked perfectly predictable and nothing — surprise,
// flourishing, seeking — could ever move. One shift was the difference
// between a living mind and a flat one.)
self.free_energy = (fe / N_SOMATIC as i32).clamp(0, i16::MAX as i32) as i16;
self.free_energy
}
}

#[inline]
fn q88_div_safe(a: i16, b: i16) -> i16 {
if b == 0 {
return 0;
}
(((a as i32) << 8) / b as i32).clamp(i16::MIN as i32, i16::MAX as i32) as i16
}
